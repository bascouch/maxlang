SCENE 04_georg
#GEORG-B


# filter+
CHAIN filtered
in adc=[1,2]
filter name=filterseqD freq=3000 mode=2 width=12 gain=8
reverb size=50 decay=50 width=100 drywet=50
out vca=1 dac=[1,2] gain=5

# glouglou-trans
CHAIN glouglou-trans
in adc=[1,2]
svptrans name=svp1 transp=300. transpadd=rand(freq=2.4 min=0. max=10.) envscale=rand(min=0.4 max=0.6 freq=4.5)  sinus=1. error=0.6 envpres=1
reverb size=40 decay=60 drywet=60
out vca=2 dac=[1,2] gain=4

# delay 2
CHAIN delay_two
in adc=[1,2]
delay name=delayseqC time=10000 gain=3 feedback=15. gain=input(in=pedal1 min=-90. max=6. curve=-0.5)
freqshift name=shishi shift=randi(freq=5 min=-0 max=200 curve=0.3)
reverb size=70 decay=50 drywet=50
out vca=3 dac=[1,2] gain=6

# sfplayer
CHAIN sfplayer
sfplayer open=s01Georg.m4a play=1 loop=1 gain=lfo(freq=3 min=-6 max=0)
reverb size=80 decay=50 width=70 drywet=50
out vca=7 dac=[1,2] gain=-13

# sfplayer-bis
CHAIN sfplayerbis
sfplayer open=s04GreysGazeX.wav play=1 loop=1 gain=lfo(freq=0.6 min=-4 max=1)
out vca=8 dac=[1,2] gain=-6
