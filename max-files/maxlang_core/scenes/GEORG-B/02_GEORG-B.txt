SCENE 02_georg
#GEORG-B

# spring
CHAIN spring
in adc=[1,2] # input Clar
delay time=rand(freq=0.2 varifreq=1. min=0. max=800) feedback=50
reverb size=80 decay=100
out vca=1 dac=[1,2] gain=6

# additive reverb
CHAIN additiveitiv
in adc=[1,2] # input Clar
filter freq=50 gain=3 mode=3
additive name=addi2 freqtransp=randi(freq=0.5 varifreq=0.4 mode=2 min=-14 max=22) attack=10 decay=300 freqsmoo=2 npeak=8 filteramp=-50 gain=input(in=pedal1, min=-90., max=6.,curve=-0.2)
reverb decay=110
out vca=2 dac=[1,2] gain=8

# twist
CHAIN twist
in adc=[1,2] # input Clar
gizmo ratio=42. gain=input(in=pedal1, min=-40. max=10. curve=-0.2)
reverb size= 90 decay=80 drywet=70
out vca=3 dac=[1,2] gain=5

# sfplayer
CHAIN sfplayer
sfplayer name=sf1 open=s01quiver.wav play=1 loop=1 gain=lfo(freq=5 min=-10 max=0)
out vca=7 dac=[1,2] gain=-5

# sfplayer
CHAIN sfplayerbis
sfplayer name=sf2 open=s01quiverX.wav play=1 loop=1 gain=lfo(freq=0.2 min=-9 max=1)
out vca=8 dac=[1,2] gain=-5
