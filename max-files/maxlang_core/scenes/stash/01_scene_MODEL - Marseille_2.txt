SCENE marseille2

# pitch
CHAIN pitchshift
in adc=[1,2] # input Clar
pitchshift transp=  lfo(freq=1 min=-5 max=5)
out


# delay
CHAIN delay
in adc=[1,2] # input Clar
delay time=rand(freq=2 min=100,max=2000) feedback=60 drywet=50 gain=rand(min=-1 max=4)
out dac=[1,2,3,4]

# sfplayer
CHAIN sfplayer
sfplayer name=player1 gain=4 open=s48_P_P_D_GLOCK_long_13A_2.aif play=1 loop=1 speed=0.1
disto drive=0.9 offset=1. gain=0
out vca=1



################
#### EVENTS ####
################

EVENTS

ONSTART # when scene is enabled



ONRELEASE



# 1
CUE pedal1
marseille2.pitchshift fadein time=100. curve=1
