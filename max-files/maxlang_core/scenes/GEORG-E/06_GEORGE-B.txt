# multidelay-gran
CHAIN multidelay-gran
in adc=[1,2] # input Clar
multidelay ingain=9 time=500 timeframe=6000 timecurve=2 ampdecay=7 feedback=20 drywet=50
granul density=8 density-jit=2 freeze=0 spat-jit=0.5
reverb size=90 decay=80 drywet=70
out vca=1 dac=[1,2] gain=6

CHAIN spec
in adc=[1,2]

svpring  name=ring drywet=50 scrubmode=1 scrub=randi(name=scrubrand min=0 max=5000 freq=0.15) feedback=80
#multidelay drywet=50 time=100 timeframe=2000 ampdecay=1 seed=CBA
out vca=2 dac=[1,2] gain=10

# spring
CHAIN springshift
in

delay time=rand(freq=0.2 varifreq=1. min=0. max=800) feedback=20

reverb size=80 decay=100
filter mode=3 freq=160 width=3
out vca=3 dac=[1,2] gain=3

# additive reverb
CHAIN mdelay-add
in
filter freq=50

disto drive=0.1 offset=0. gain=-12
reverb decay=110
out vca=4 dac=[1,2] gain=3
