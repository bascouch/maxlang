SCENE 01_georg
#GEORG-D EFFUVES D'AVRIL I

# delay 1
CHAIN delay_one
in adc=[1,2]
delay name=delayseqC time=rand(freq=0.24 min=0. max=200 varifreq=0.5) feedback=15. gain=input(in=pedal1 min=-40. max=6. curve=-0.2)
imager rotation=40
reverb size=50 decay=100 drywet=80
out vca=1 dac=[1,2] gain=-1

# delay 2
CHAIN delay_two
in adc=[1,2]
delay name=delayseqC time=rand(freq=0.34 min=0. max=400 varifreq=0.6) feedback=25. gain=3
imager rotation=35
reverb size=70 decay=120 drywet=80
out vca=2 dac=[1,2] gain=2

# twist
CHAIN twist
in adc=[1,2] # input Clar
gizmo ratio=42. gain=input(in=pedal1 min=-40. max=10. curve=-0.2)
imager rotation=randi(freq=0.5 min=-45 max=45)
reverb size= 90 decay=80 drywet=70
out vca=3 dac=[1,2] gain=8

# sfplayer
CHAIN sfplayer-twist
sfplayer name=sft open=NEW_KIDS_4.wav play=1 loop=0 gain=2
imager rotation=-35
out vca=5 dac=[1,2] gain=-6

# sfplayer
CHAIN sfplayer
sfplayer name=sf1 open=SHORT_GREYS_A_7.wav play=1 loop=0 gain=lfo(freq=0.4 min=-6 max=2)
imager rotation=-40
out vca=7 dac=[1,2] gain=1
#gain=lfo(freq=5 min=-10 max=0)

# sfplayer-bis
CHAIN sfplayerbis
sfplayer name=sf2 open=SHORT_GREYS_B_7.wav play=1 loop=0 gain=lfo(freq=0.9 min=-7 max=1)
imager rotation=40
out vca=8 dac=[1,2] gain=4
#gain=lfo(freq=0.2 min=-9 max=1)
