SCENE myscene1

# pitch
CHAIN mychain1
in
additive name=add npeak=2 attack=5 decay=5 freqsmoo=5 hrm=0.2 randdev=20 freqtransp=-24 drywet=89
freqshift name=shi shift=0.1
reverb
out vca=1 dac=[1 2 4 10]

# pitch
CHAIN mychain2-psycho
in
psychoirtrist voices=1 select=lfo(min=1 max=16 freq=0.03) transp-list=[@nrand(4 -1200 1200)] transp=randi(min=-200 max=200 freq=0.2) delay=0 tempmod=300
reverb decay=100
multidelay drywet=50 feedback=50 time=100 timeframe=1000 ampdecay=0 seed=CBA
spat rotation=lfo(min=0 max=360 freq=4) width=1 vca=1 dac=[1 2]

CHAIN file

sfplayer open=Rec-20.05.11-09h28m59s.aif loop=1 speed=0.5 play=1 gain=0

imager rotation=lfo(min=-45. max=20. freq=0.1 varifreq=2) balance=0.5 width=1.
reverb drywet=60
out dac=[1 2] vca=2
