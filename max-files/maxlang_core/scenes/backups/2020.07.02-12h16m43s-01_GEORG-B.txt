SCENE 01_georg
#GEORG-B

# svptransouf
CHAIN svptransouf
in adc=[1,2] # input Clar
delay time=0
svptrans name=svp1 transp=rand(freq=0.4 min=-25. max=25.) envscale=lfo(freq=0.02 min=0.6 max=1. curve=0.5) sinus=1. error=0.2 gain=0
reverb size=30 decay=75 damping=30 drywet=20
filter name=outfilter mode=2 freq=1800 width=90
out vca=1 dac=[1,2] gain=lfo(mode=2 freq=randi(min=3 max=12 freq=1) min=-18. max=4. pw=0.5)

#pitch
CHAIN bigdisto
in adc=[1,2] # input Clar
disto gain=input(in=pedal1, min=-90. max=6. curve=-0.2)
filter name=filterseqC freq=rand(min=50 max=200 freq=0.1 curve=0.2) mode=3 width=7
delay name=delayseqC time=randi(freq=5 varifreq=1 walk=0.5 min=0 max=1000 curve=0.6) drywet=30 feedback=70
freqshift name=shishi shift=randi(name=gigilfo,freq=6 min=-70 max=50 curve=0)
reverb size=100 decay=85 drywet=60
out vca=2 dac=[1,2] gain=6

# granulation
CHAIN granulle
in adc=[1,2]
granul density=13 density-jit=0.5 length=10 delay=rand(min=400 max=1000) delay-mod-depth=500 gain=input(in=pedal1, min=-90., max=6.,curve=-0.2)
filter mode=1 freq=200
reverb size=90 decay=90 drywet=30
out vca=3 dac=[1,2] gain=-3

# shiftsimple
CHAIN shiftsimple
in adc=[1,2]
freqshift shift=randi(min=-1000,max=-800,freq=0.1)
reverb size=80 decay=100
out vca=4 dac=[1,2] gain=9

# sfplayer
CHAIN sfplayer
sfplayer name=arpplayer open=arp2.aif loop=0 gain=lfo(freq=8 min=-7 max=4) play=1
out vca=8


################
#### EVENTS ####

EVENTS

ONSTART # when scene is enabled



## Quand on quitte la scene
# extinction de toutes les chaines

ONRELEASE
01_georg.svptransouf fadeoutmute time=7000

01_georg.granulle fadeoutmute time=7000
01_georg.shiftsimple fadeoutmute time=7000
01_georg.sfplayer fadeoutmute time=7000

CUE pedal1

# 2
CUE pedal2

# 3
CUE pedal3

01_georg.bigdisto fadein time=1000. curve=1
# 4
CUE pedal4
01_georg.bigdisto fadeout time=7000

# 5
CUE pedal5
01_georg.granulle fadein time=1000
# 6
CUE pedal6
01_georg.granulle fadeout time=7000

# 7
CUE pedal7
01_georg.sfplayer fadein time=3000 curve=-0.8
01_georg.sfplayer.arpplayer play 1
01_georg.shiftsimple fadein time=3000 curve=-0.8
#  8
CUE pedal8
01_georg.shiftsimple fadeout time=7000
01_georg.sfplayer fadeout time=7000
