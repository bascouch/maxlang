SCENE scene1

# pitch
CHAIN transpolfofo
in adc=[1,2] # input Clar
filter name=filterseqC freq=220 mode=3 width=12
delay name=delayseqC time=rand(freq=3,varifreq=0.5 ,min=0,max=2000,curve=2)
freqshift name=shishi shift=randi(name=gigilfo,freq=lfo(freq=0.5,min=0.05,max=0.2), min=-100, max=100, curve=0)
reverb size=120 decay=75 drywet=50
out vca=1 dac=[1,2,3,4]

# additive reverb
CHAIN additive
in
filter freq=50
additive name=addi2 freqtransp=randi(freq=1.5,varifreq=0.2,mode=2,min=-36,max=12) attack=10 decay=300 freqsmoo=2 npeak=8 filteramp=-50 gain=midictl(num=86, min=-6., max=12.)
reverb decay=110
out vca=2 dac=[1,2] gain=3

#CHAIN
CHAIN svptrans
in
delay time=0
svptrans name=svp1 transp=rand(freq=0.4,min=0.,max=10.) envscale=lfo(freq=0.02,min=0.6,max=1.7,curve=0.1) sinus=1. error=0.2 gain=0
reverb size=30 decay=75 damping=30 drywet=20
filter name=outfilter mode=2 freq=1800 width=90
out vca=3

CHAIN shiftsimple
in
receivebus bus=tramecrotale
freqshift shift=randi(min=-1000,max=-800,freq=0.1)
reverb size=80 decay=100
out vca=7

# SOUNDFILE
CHAIN soundfile_Trame
sfplayer name=player1 open=s48_P_P_A_TRAME_4A-2.wav loop=0 speed=1
sendbus bus=tramecrotale
out vca=8


################
#### EVENTS ####
################

EVENTS

ONSTART # when scene is enabled
transpolfofo mute 1


ONRELEASE
scene1.transpolfofo fadeoutmute time=3000
scene1.additive fadeoutmute time=3000
scene1.additive&spat fadeoutmute time=3000


# 1
CUE pedal1
scene1.transpolfofo enable 1
scene1.transpolfofo fadein time=100. curve=1


# 2
CUE pedal2
scene1.transpolfofo fadeoutmute time=2000 curve=1

# 3
CUE pedal3
scene1.additive enable 1
scene1.additive fadein time=1000. curve=1

# 4
CUE pedal4
scene1.additive fadeout time=2000

# 5
CUE pedal5
scene1.svptrans enable 1
scene1.svptrans fadein time=100

# 6
CUE pedal6
scene1.svptrans fadeout

# 7
CUE pedal7
scene1.shiftsimple enable 1
scene1.shiftsimple fadein

#  8
CUE pedal8
scene1.shiftsimple fadeout
