SCENE ANUSH

# pitch
CHAIN dry
in adc=[1 1]
out vca=1

# pitch
CHAIN additive
in adc=[1 1]
additive npeak=2 decay=100 freqtransp=12 hrm=0.2
reverb
out vca=2

# additive reverb
CHAIN file
sfplayer open=s04GreysGazeX.wav loop=1 play=1 gain=6
#spectralcompand sinusnoise=0 threshold=lfo(min=-60. max=-20. freq=0.2)
sendbus bus=file
imager rotation=randi(min=-90 max=90 freq=0.5) width=1 balance=0
out gain=0

CHAIN file_side
receivebus bus=file
filter freq=2000 mode=2 width=12 gain=24
imager rotation=randi(min=-90 max=90 freq=0.5) width=1 balance=0
out vca=2
