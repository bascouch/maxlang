SCENE 01_chaumont

#pitch
CHAIN transpolfofo
in adc=[1,2] # input Clar
filter name=filterseqC freq=randi(min=50 max=200 freq=0.1 curve=0.2) mode=3 width=7
delay name=delayseqC time=rand(freq=3 varifreq=0.5 min=0 max=2000 curve=0.6) drywet=30 feedback=70
freqshift name=shishi shift=randi(name=gigilfo,freq=6, min=-100, max=100, curve=0)
reverb size=120 decay=75 drywet=50
out vca=1 dac=[1,2] gain=0

# additive reverb
CHAIN additiveitiv
in
filter freq=50
additive name=addi2 freqtransp=-17 attack=500 decay=1000 freqsmoo=1000 npeak=2 filteramp=-50 gain=input(in=pedal1  min=-90.  max=6. curve=-0.2)
disto drive=0.9 offset=0. gain=-12
filter freq=400 mode=1 width=12
reverb decay=100
out vca=2 dac=[1,2] gain=0

# svptransouf
CHAIN svptransouf
in
delay time=0
svptrans name=svp1 transp=rand(freq=0.4,min=0.,max=10.) envscale=lfo(freq=0.02,min=0.6,max=1.7,curve=0.05) sinus=1. error=0.2 gain=0
reverb size=30 decay=75 damping=30 drywet=20
filter name=outfilter mode=2 freq=1800 width=90
out vca=3 dac=[1,2] gain=lfo(mode=2 freq=randi(min=3 max=12 freq=1) min=-18. max=4. pw=0.5)

# shiftsimple
CHAIN shiftsimple
in
receivebus bus=tramecrotale
freqshift shift=randi(min=-1000,max=-800,freq=0.1)
reverb size=80 decay=100
out vca=4 dac=[1,2] gain=8

# sfplayer
CHAIN sfplayer
sfplayer open=48_P_P_A_TRAME_4A-2.wav loop=0 gain=lfo(freq=5 min=-10 max=0)
out vca=8
