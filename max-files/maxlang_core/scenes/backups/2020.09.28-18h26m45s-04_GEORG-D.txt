# psychochoristo

in adc=[1,2]

#filter+
CHAIN filtered
in adc=[1,2]

imager rotation=lfo(mode=2 freq=0.15 min=0. max=160.)
freqshift shift=-400
reverb size=50 decay=50 width=100 drywet=50
out vca=2 dac=[1,2] gain=11
# dry part of pedal1
CHAIN filtered-dry
in adc=[1,2]
imager rotation=10
reverb size=50 decay=50 width=100 drywet=50
out vca=2 dac=[1,2] gain=6

# spec-test
CHAIN spec-test
in adc=[1,2] # input Clar
spectralcompand name=spcomp sinusnoise=100
svpring  name=ring drywet=50 scrubmode=1 scrub=randi(name=scrubrand min=0 max=5000 freq=0.15)
multidelay drywet=50 time=100 timeframe=5000 ampdecay=6 seed=CBA
imager rotation=lfo(mode=3 freq=0.05 min=0.4 max=260.)
#filter mode=0 freq=9000
out vca=3 dac=[1,2] gain=13
